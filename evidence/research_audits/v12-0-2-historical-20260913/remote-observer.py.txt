def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=False,
                      allow_nan=False).encode('utf-8')


def sha256(raw):
    return hashlib.sha256(raw).hexdigest()


def get(path):
    if not path.startswith('/repos/Jasper-Shi/lottopred/'):
        raise SystemExit('Unregistered read endpoint.')
    proc = subprocess.run([GH, 'api', '--hostname', 'github.com', '--method', 'GET', path],
                          capture_output=True, check=False)
    if proc.returncode:
        raise SystemExit('GET-only remote observation unavailable; no remote mutation made.')
    return json.loads(proc.stdout)


GIT_ENV = {
    'PATH': '/usr/bin:/bin:/usr/sbin:/sbin', 'HOME': '/private/tmp', 'LANG': 'C',
    'GIT_CONFIG_NOSYSTEM': '1', 'GIT_CONFIG_GLOBAL': '/dev/null',
    'GIT_NO_REPLACE_OBJECTS': '1', 'GIT_TERMINAL_PROMPT': '0',
    'GIT_OPTIONAL_LOCKS': '0',
}


def git(*args):
    proc = subprocess.run(['/usr/bin/git', '-c', 'core.hooksPath=/dev/null',
                           '-c', 'core.fsmonitor=false', '-C', str(ROOT), *args],
                          env=GIT_ENV, capture_output=True, check=False)
    if proc.returncode:
        raise SystemExit('Read-only local Git identity check failed.')
    return proc.stdout


def ref_check(value, ref, oid):
    assert set(value) == {'ref', 'node_id', 'url', 'object'}
    assert isinstance(value['node_id'], str) and value['node_id']
    assert value['ref'] == ref
    assert value['url'] == API + '/git/' + ref
    assert value['object'] == {'type': 'commit', 'sha': oid, 'url': API + '/git/commits/' + oid}
