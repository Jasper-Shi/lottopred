class AuditFailure(ValueError):
    """A fixed input or independently recomputed result did not match."""


def require(condition, message):
    if not condition:
        raise AuditFailure(message)


def close(actual, expected, message):
    require(
        type(actual) in (int, float)
        and type(expected) in (int, float)
        and math.isfinite(actual)
        and math.isfinite(expected)
        and math.isclose(actual, expected, rel_tol=2e-12, abs_tol=2e-12),
        message,
    )


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def canonical(value):
    return (
        json.dumps(
            value,
            sort_keys=True,
            separators=(",", ":"),
            ensure_ascii=False,
            allow_nan=False,
        )
        + "\n"
    ).encode()


def strict_json(raw):
    def pairs(items):
        result = {}
        for key, value in items:
            require(key not in result, "duplicate JSON key")
            result[key] = value
        return result

    def reject_constant(value):
        raise AuditFailure("nonfinite JSON constant: " + value)

    return json.loads(raw, object_pairs_hook=pairs, parse_constant=reject_constant)


def calendar():
    current, end, dates = date(2020, 1, 1), date(2025, 12, 31), []
    while current <= end:
        if current.weekday() in (2, 5):
            dates.append(current.isoformat())
        current += timedelta(days=1)
    return dates


def date_hash(dates):
    return sha(("\n".join(dates) + "\n").encode())

def readonly_guard(event, args):
    """Defense in depth for this Python process; not an OS sandbox claim."""
    if event == "open":
        _, mode, flags = args
        if (mode and any(letter in mode for letter in "wax+")) or (
            flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT | os.O_TRUNC | os.O_APPEND)
        ):
            raise AuditFailure("read-only replay rejected a file write")
    if event.startswith("socket."):
        raise AuditFailure("read-only replay rejected network access")
    if event == "import" and (
        args[0].startswith("lotto649.models")
        or "registered_attempt" in args[0]
        or args[0] in ("lotto649.notification", "lotto649.v12_0_2_evidence")
    ):
        raise AuditFailure("replay rejected a forecasting/worker module import")
    if event == "subprocess.Popen":
        executable, argv, _, _ = args
        require(executable in ("git", "/usr/bin/git"), "only local Git reads allowed")
        verbs = {
            "show",
            "rev-list",
            "rev-parse",
            "ls-tree",
            "cat-file",
            "merge-base",
            "diff-tree",
            "diff",
        }
        index = 1
        while index < len(argv) and argv[index].startswith("-"):
            option = argv[index]
            require(option in ("-C", "-c", "--no-replace-objects"), "Git global option")
            index += 2 if option in ("-C", "-c") else 1
        require(index < len(argv) and argv[index] in verbs, "Git write/network verb")

def exact_laws(sample_sizes):
    require(
        sample_sizes and all(type(n) is int and n > 0 for n in sample_sizes),
        "positive sample sizes required",
    )
    weights = [math.comb(12, k) * math.comb(37, 6 - k) for k in range(7)]
    require(sum(weights) == N, "single-draw exact-law mass")
    coefficients, laws = [1], {}
    for size in range(1, max(sample_sizes) + 1):
        updated = [0] * (len(coefficients) + 6)
        for hits, weight in enumerate(weights):
            for total, coefficient in enumerate(coefficients):
                updated[total + hits] += weight * coefficient
        coefficients = updated
        if size in sample_sizes:
            require(sum(coefficients) == N**size, "convolution exact mass")
            laws[size] = coefficients
    return laws


def exact_tail(law, size, total):
    require(type(total) is int and 0 <= total <= size * 6, "exact tail domain")
    numerator, denominator = sum(law[total:]), N**size
    return {
        "numerator_hex": hex(numerator),
        "denominator_hex": hex(denominator),
        "value": numerator / denominator,
    }


def bootstrap(np, values, indices=None):
    require(
        values and all(type(x) in (int, float) and math.isfinite(x) for x in values),
        "bootstrap finite nonempty array",
    )
    n = len(values)
    if indices is None:
        indices = np.random.default_rng(649).integers(0, n, size=(10000, n))
    means = np.array(
        [math.fsum(values[int(i)] for i in sample) / n for sample in indices],
        dtype=np.float64,
    )
    lower, upper = np.quantile(means, [0.025, 0.975], method="linear")
    return {
        "lower": float(lower),
        "upper": float(upper),
        "seed": 649,
        "resamples": 10000,
        "method": "linear",
    }


def check_interval(calculated, recorded, label):
    require(set(recorded) == set(calculated), label + " keys")
    for key in ("seed", "resamples", "method"):
        require(recorded[key] == calculated[key], label + " " + key)
    for key in ("lower", "upper"):
        close(recorded[key], calculated[key], label + " " + key)

def check_model_summary(rows, index, summary):
    scores = [row["scores"][index] for row in rows]
    require(
        summary["draw_count"] == len(rows) and summary["model_name"] == MODELS[index],
        "model summary identity",
    )
    for source, target in (
        ("brier_score", "brier_score"),
        ("log_loss", "log_loss"),
        ("mean_actual_rank", "mean_actual_rank"),
        ("final6_hits", "mean_final6_hits"),
    ):
        close(
            summary[target],
            math.fsum(s[source] for s in scores) / len(rows),
            "model mean " + target,
        )
    for k in (6, 12, 18):
        mean = math.fsum(s[f"top{k}_hits"] for s in scores) / len(rows)
        close(summary[f"mean_top{k}_hits"], mean, "mean Top-K hits")
        close(summary[f"top{k}_lift"], mean - 6 * k / 49, "fair Top-K lift")
    require(
        summary["final6_histogram"]
        == {str(k): sum(s["final6_hits"] == k for s in scores) for k in range(7)},
        "complete Final-6 histogram",
    )
    if index < 2:
        close(
            summary["joint_log_gain_sum"],
            math.fsum(s["joint_log_gain"] for s in scores),
            "joint score sum",
        )
    else:
        require(
            summary["joint_log_gain_sum"] is None, "joint score absent for comparator"
        )
    bins = [[] for _ in range(10)]
    for row in rows:
        probabilities = row["forecast_payload"]["forecasts"][index]["probabilities"]
        for label in range(1, 50):
            p = probabilities[str(label)]
            bucket = next(i for i in range(10) if i / 10 <= p < (i + 1) / 10)
            bins[bucket].append((p, int(label in row["actual"])))
    require(len(summary["calibration_bins"]) == 10, "ten calibration bins")
    for i, values in enumerate(bins):
        recorded = summary["calibration_bins"][i]
        require(
            recorded["lower"] == i / 10
            and recorded["upper"] == (i + 1) / 10
            and recorded["count"] == len(values),
            "calibration bin bounds/count",
        )
        if values:
            close(
                recorded["mean_probability"],
                math.fsum(p for p, _ in values) / len(values),
                "calibration mean probability",
            )
            close(
                recorded["observed_frequency"],
                math.fsum(y for _, y in values) / len(values),
                "calibration observed frequency",
            )
        else:
            require(
                recorded["mean_probability"] is None
                and recorded["observed_frequency"] is None,
                "empty calibration bins retained",
            )


def replay_statistics(np, report):
    rows = report["targets"]
    scopes = {"aggregate": rows, "first_half": rows[:314], "second_half": rows[314:]}
    laws = exact_laws({313, 314, 627})
    intervals, exact_tests = [], []
    for scope_name, scope_rows in scopes.items():
        recorded = report["scopes"][scope_name]
        dates, n = [r["target_draw_date"] for r in scope_rows], len(scope_rows)
        require(
            recorded["n"] == n
            and recorded["first_target"] == dates[0]
            and recorded["last_target"] == dates[-1]
            and recorded["target_dates_sha256"]
            == date_hash(dates)
            == DATE_HASHES[scope_name],
            "scope date membership",
        )
        # Every contrast logically starts a fresh seed-649 generator. Reusing its
        # identical index matrix here preserves every sampled complete row exactly.
        indices = np.random.default_rng(649).integers(0, n, size=(10000, n))
        for index, model in enumerate(MODELS):
            summary = recorded["models"][model]
            check_model_summary(scope_rows, index, summary)
            hits = [row["scores"][index]["top12_hits"] for row in scope_rows]
            exact = exact_tail(laws[n], n, sum(hits))
            require(
                exact["numerator_hex"] == summary["top12_exact_p"]["numerator_hex"]
                and exact["denominator_hex"]
                == summary["top12_exact_p"]["denominator_hex"],
                "exact unreduced integer tail ratio",
            )
            close(exact["value"], summary["top12_exact_p"]["value"], "exact tail float")
            ci = bootstrap(np, [h - FAIR_TOP12 for h in hits], indices)
            check_interval(ci, summary["top12_lift_ci95"], "Top-12 lift CI")
            intervals.append({"scope": scope_name, "model_or_contrast": model, **ci})
            exact_tests.append(
                {
                    "scope": scope_name,
                    "model": model,
                    "n": n,
                    "total_top12_hits": sum(hits),
                    "value": exact["value"],
                    "exact_ratio_sha256": sha(canonical(exact)),
                }
            )
        for name, index in (("candidate_minus_v1", 2), ("candidate_minus_pseudo", 1)):
            ci = bootstrap(
                np,
                [
                    row["scores"][0]["top12_hits"] - row["scores"][index]["top12_hits"]
                    for row in scope_rows
                ],
                indices,
            )
            contrast = recorded["contrasts"][name]
            check_interval(ci, contrast["top12_paired_ci95"], "paired Top-12 CI")
            intervals.append({"scope": scope_name, "model_or_contrast": name, **ci})
            for metric, key in (
                ("brier_score", "brier_delta"),
                ("log_loss", "log_loss_delta"),
            ):
                delta = (
                    math.fsum(
                        row["scores"][0][metric] - row["scores"][index][metric]
                        for row in scope_rows
                    )
                    / n
                )
                close(contrast[key], delta, "paired proper-score contrast")
            if index == 1:
                close(
                    contrast["joint_log_gain_sum"],
                    math.fsum(
                        row["scores"][0]["joint_log_gain"]
                        - row["scores"][1]["joint_log_gain"]
                        for row in scope_rows
                    ),
                    "paired joint score sum",
                )
        for metric, key in (
            ("brier_score", "brier_delta"),
            ("log_loss", "log_loss_delta"),
        ):
            fair = (
                FAIR * (1 - FAIR)
                if metric == "brier_score"
                else (-FAIR * math.log(FAIR) - (1 - FAIR) * math.log1p(-FAIR))
            )
            close(
                recorded["contrasts"]["candidate_minus_fair"][key],
                math.fsum(row["scores"][0][metric] - fair for row in scope_rows) / n,
                "fair proper-score contrast",
            )
    require(
        len(intervals) == 18 and len(exact_tests) == 12,
        "all registered inference replays",
    )
    best = {}
    for index, model in enumerate(MODELS):
        maximum = max(row["scores"][index]["final6_hits"] for row in rows)
        best[model] = {
            "final6_hits": maximum,
            "target_dates": [
                row["target_draw_date"]
                for row in rows
                if row["scores"][index]["final6_hits"] == maximum
            ],
        }
    require(
        best == report["best_final6_by_model"], "all tied best dates, no cherry-picking"
    )
    return {
        "confidence_intervals": intervals,
        "exact_tail_ratios": exact_tests,
        "all_best_dates": best,
    }


def self_test(np):
    passed = []

    def rejects(label, operation):
        try:
            operation()
        except (AuditFailure, ValueError):
            passed.append(label)
        else:
            raise AuditFailure("negative test accepted: " + label)

    rejects("duplicate JSON", lambda: strict_json('{"a":1,"a":2}'))
    for value in ("NaN", "Infinity", "-Infinity"):
        rejects("nonfinite JSON " + value, lambda value=value: strict_json(value))
    require(
        strict_json(canonical({"b": 2, "a": 1})) == {"a": 1, "b": 2},
        "canonical round trip",
    )
    require(
        sha(canonical({"a": 1})) != sha(canonical({"a": 1}).rstrip(b"\n")), "LF binding"
    )
    passed.extend(("canonical round trip", "canonical newline hash binding"))
    dates = calendar()
    require(
        len(dates) == 627 and len(dates[:314]) == 314 and len(dates[314:]) == 313,
        "synthetic calendar split",
    )
    for key, value in (
        ("aggregate", dates),
        ("first_half", dates[:314]),
        ("second_half", dates[314:]),
    ):
        require(date_hash(value) == DATE_HASHES[key], "date digest " + key)
        passed.append("calendar/split digest " + key)
    records = [
        {"draw_date": "2019-12-28", "numbers": [1, 2, 3, 4, 5, 6], "bonus": 7},
        {"draw_date": "2020-01-01", "numbers": [8, 9, 10, 11, 12, 13], "bonus": 14},
    ]
    payload = {
        "target_draw_date": "2020-01-01",
        "history_through": "2019-12-28",
        "training_cutoff_date": "2019-12-28",
        "visible_prefix_draw_count": 1,
        "visible_prefix_sha256": sha(canonical(records[:1])),
    }
    row = {
        "target_draw_date": "2020-01-01",
        "actual": records[1]["numbers"],
        "bonus": 14,
        "forecast_payload": payload,
        "forecast_payload_sha256": sha(canonical(payload)),
    }
    authenticate_row(row, records, 1)
    passed.append("invented history prefix authentication")
    for key, value in (
        ("training_cutoff_date", "2020-01-01"),
        ("visible_prefix_draw_count", 2),
        ("visible_prefix_sha256", "0" * 64),
    ):
        mutation = copy.deepcopy(row)
        mutation["forecast_payload"][key] = value
        mutation["forecast_payload_sha256"] = sha(
            canonical(mutation["forecast_payload"])
        )
        rejects(
            "bad prefix " + key,
            lambda mutation=mutation: authenticate_row(mutation, records, 1),
        )
    mutation = copy.deepcopy(row)
    mutation["actual"][0] = 49
    rejects("altered target answer", lambda: authenticate_row(mutation, records, 1))
    laws = exact_laws({1, 2})
    require(
        laws[1] == [math.comb(12, k) * math.comb(37, 6 - k) for k in range(7)],
        "one-draw oracle",
    )
    require(sum(laws[2]) == N**2, "two-draw mass")
    require(
        exact_tail(laws[1], 1, 6)["numerator_hex"] == hex(math.comb(12, 6)),
        "six-hit tail oracle",
    )
    passed.extend(
        (
            "one-draw hypergeometric oracle",
            "two-draw exact mass",
            "six-hit exact numerator",
        )
    )
    rejects("exact tail out of range", lambda: exact_tail(laws[1], 1, 7))
    for value in (0.0, 2.5):
        ci = bootstrap(np, [value] * 4)
        require(ci["lower"] == ci["upper"] == value, "constant bootstrap oracle")
        passed.append("constant bootstrap oracle " + str(value))
    rejects("empty bootstrap", lambda: bootstrap(np, []))
    rejects("nonfinite bootstrap", lambda: bootstrap(np, [float("nan")]))
    ci = bootstrap(np, [1, 2, 4, 0])
    check_interval(ci, ci, "positive CI check")
    mutation = dict(ci, lower=ci["lower"] + 0.01)
    rejects("changed CI bound", lambda: check_interval(ci, mutation, "mutated CI"))
    rejects(
        "changed CI seed",
        lambda: check_interval(ci, dict(ci, seed=650), "mutated seed"),
    )
    return {
        "status": "PASS_SYNTHETIC_ONLY",
        "check_count": len(passed),
        "checks": passed,
        "provenance": "new post-audit reproducibility helper; not original inline audit source",
    }
